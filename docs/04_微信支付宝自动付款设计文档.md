## 一、当前状态

- **V1 版本（当前）**：HTML 付费引导页（展示二维码 → 用户手动付款 → 作者手动发授权码）
- **V2 版本（未来）**：微信支付统一下单 + 自动查询订单状态 + 付款成功后自动生成激活授权码

## 二、你需要准备的材料

### 微信支付商户平台

| 材料名称 | 说明 | 在哪里找 |
|---------|------|---------|
| **商户号 (mch_id)** | 微信支付分配的商户号，格式如 160000xxxx | 商户平台 → 账户中心 → 商户信息 |
| **API 密钥 (api_key)** | 32位字符串，用于签名 | 商户平台 → 账户中心 → API安全 → 设置密钥 |
| **API 证书** | apiclient_cert.pem + apiclient_key.pem | 商户平台 → 账户中心 → API安全 → 下载证书 |
| **商户公众号 AppID** | 你已有的微信公众号 AppID | 公众号后台 → 开发 → 基本配置 |

### 支付宝开放平台（如果做支付宝）

| 材料名称 | 说明 |
|---------|------|
| 应用 ID (app_id) | 开放平台创建应用后分配 |
| 应用私钥 (app_private_key) | 2048位 RSA 私钥 |
| 支付宝公钥 (alipay_public_key) | 开放平台对应 |

### 4.2 core/payment_wechat.py 微信支付模块

（代码框架，无需现在运行）

```python
"""
微信支付自动模块 V2 框架
对接 微信支付 V3 API
需要准备：appid, mch_id, api_key, 证书文件
"""

import hashlib
import random
import time
import json
import requests
import xml.etree.ElementTree as ET
from datetime import datetime
from pathlib import Path


class WechatPayClient:
    """微信支付客户端 - 统一下单 + 订单查询 + 自动授权"""

    def __init__(self, config: dict):
        self.appid = config["appid"]
        self.mch_id = config["mch_id"]
        self.api_key = config["api_key"]
        self.cert_path = config.get("cert_path", "")
        self.key_path = config.get("key_path", "")
        self.enabled = config.get("enabled", False)

    # ========== 步骤1：生成统一下单，返回支付二维码链接 ==========
    def create_order(
        self,
        product_id: str,
        product_name: str,
        amount_fen: int,
        user_machine_code: str,
    ) -> dict:
        """
        创建 Native 扫码支付订单

        参数:
            product_id: 商品ID（如 'member_month_888'）
            product_name: 商品名称（如 '普通会员包月-8.88元'）
            amount_fen: 金额（单位：分，8.88元 = 888分）
            user_machine_code: 用户机器码，用于订单关联

        返回:
            {
                "success": True,
                "out_trade_no": "订单号",
                "code_url": "weixin://wxpay/bizpayurl?pr=xxxx",  // 用于生成二维码
                "amount": 8.88,
                "expire_seconds": 600,
            }
        """
        if not self.enabled:
            return {"success": False, "error": "微信支付未启用，请使用扫码付款手动激活"}

        out_trade_no = self._gen_out_trade_no(user_machine_code)

        # 组装参数
        params = {
            "appid": self.appid,
            "mch_id": self.mch_id,
            "nonce_str": self._gen_nonce_str(),
            "body": product_name,
            "out_trade_no": out_trade_no,
            "total_fee": amount_fen,
            "spbill_create_ip": "127.0.0.1",
            "notify_url": "",  # 回调URL（纯CMD版可不用，靠轮询查询订单）
            "trade_type": "NATIVE",
            "product_id": product_id,
            "attach": json.dumps({"machine_code": user_machine_code}),
        }
        params["sign"] = self._sign(params)

        # 调用微信统一下单API
        try:
            resp = requests.post(
                "https://api.mch.weixin.qq.com/pay/unifiedorder",
                data=self._dict_to_xml(params),
                cert=(self.cert_path, self.key_path) if self.cert_path else None,
                timeout=15,
            )
            result = self._xml_to_dict(resp.content.decode("utf-8"))

            if result.get("return_code") == "SUCCESS" and result.get("result_code") == "SUCCESS":
                return {
                    "success": True,
                    "out_trade_no": out_trade_no,
                    "code_url": result.get("code_url", ""),
                    "amount_yuan": round(amount_fen / 100, 2),
                    "expire_seconds": 600,
                }
            else:
                return {
                    "success": False,
                    "error": result.get("return_msg", result.get("err_code_des", "下单失败")),
                }
        except Exception as e:
            return {"success": False, "error": f"网络异常: {str(e)}"}

    # ========== 步骤2：轮询查询订单状态，付款成功自动激活授权码 ==========
    def query_order_and_auto_activate(
        self,
        out_trade_no: str,
        user_machine_code: str,
        auth_manager_instance,
        max_wait_seconds: int = 300,
        interval_seconds: int = 3,
    ) -> dict:
        """
        轮询查询订单状态，付款成功后自动生成并激活授权码

        参数:
            out_trade_no: 商户订单号
            user_machine_code: 用户机器码
            auth_manager_instance: AuthManager 实例
            max_wait_seconds: 最大等待时间（秒），默认5分钟
            interval_seconds: 轮询间隔（秒）

        返回:
            {
                "success": True,
                "activated": True,
                "auth_code": "MC-XXXX-XXXX-XXXX",
                "plan_type": "member_month",
            }
        """
        if not self.enabled:
            return {"success": False, "activated": False, "error": "微信支付未启用"}

        start_time = time.time()
        while time.time() - start_time < max_wait_seconds:
            result = self._query_order(out_trade_no)
            if result.get("trade_state") == "SUCCESS":
                # 付款成功，解析商品类型，自动激活
                attach_raw = result.get("attach", "{}")
                attach = json.loads(attach_raw) if attach_raw else {}
                plan_type = attach.get("plan_type", "member_month")

                # 生成并激活授权码
                auth_code = auth_manager_instance.generate_random_auth_code(plan_type)
                activated = auth_manager_instance.activate_auth_code(auth_code, user_machine_code)

                if activated:
                    return {
                        "success": True,
                        "activated": True,
                        "auth_code": auth_code,
                        "plan_type": plan_type,
                        "message": "付款成功，已自动激活授权码",
                    }
                else:
                    return {
                        "success": True,
                        "activated": False,
                        "message": "付款成功但激活失败，请手动联系作者",
                        "transaction_id": result.get("transaction_id", ""),
                    }

            time.sleep(interval_seconds)

        return {
            "success": False,
            "activated": False,
            "timeout": True,
            "message": "等待付款超时，请检查微信支付状态",
        }

    # ========== 内部工具方法 ==========

    def _query_order(self, out_trade_no: str) -> dict:
        """查询微信订单状态"""
        params = {
            "appid": self.appid,
            "mch_id": self.mch_id,
            "out_trade_no": out_trade_no,
            "nonce_str": self._gen_nonce_str(),
        }
        params["sign"] = self._sign(params)

        try:
            resp = requests.post(
                "https://api.mch.weixin.qq.com/pay/orderquery",
                data=self._dict_to_xml(params),
                timeout=10,
            )
            return self._xml_to_dict(resp.content.decode("utf-8"))
        except Exception:
            return {}

    def _gen_out_trade_no(self, machine_code: str) -> str:
        """生成唯一商户订单号：时间戳+机器码哈希+随机数"""
        ts = datetime.now().strftime("%Y%m%d%H%M%S")
        mc_hash = hashlib.md5(machine_code.encode()).hexdigest()[:8]
        rand = random.randint(1000, 9999)
        return f"MC{ts}{mc_hash}{rand}"

    def _gen_nonce_str(self) -> str:
        """生成32位随机字符串"""
        chars = "abcdefghijklmnopqrstuvwxyz0123456789"
        return "".join(random.choices(chars, k=32))

    def _sign(self, params: dict) -> str:
        """微信签名算法 MD5"""
        sorted_items = sorted(params.items())
        sign_str = "&".join(f"{k}={v}" for k, v in sorted_items if v)
        sign_str = f"{sign_str}&key={self.api_key}"
        return hashlib.md5(sign_str.encode("utf-8")).hexdigest().upper()

    def _dict_to_xml(self, params: dict) -> str:
        xml = "<xml>"
        for k, v in params.items():
            xml += f"<{k}><![CDATA[{v}]]></{k}>"
        xml += "</xml>"
        return xml

    def _xml_to_dict(self, xml_text: str) -> dict:
        result = {}
        try:
            root = ET.fromstring(xml_text)
            for child in root:
                result[child.tag] = child.text
        except Exception:
            pass
        return result
```

## 五、升级步骤（V1 → V2）

当你准备好商户信息后，按以下步骤启用自动支付：

| 步骤 | 操作 |
|------|------|
| 1 | 把商户号、API密钥、证书文件放到 `secrets/` 目录 |
| 2 | 修改 `config.py` 中的 `WECHAT_PAY`，填上值并把 `enabled` 改为 `True` |
| 3 | 把 `core/payment_wechat.py` 从框架代码补全为可用代码（主要是调试签名和证书） |
| 4 | 修改 `main.py` 启用 `--pay` 参数 |
| 5 | 测试：用1分钱模拟订单（微信支付沙箱），验证完整付款→激活流程 |
| 6 | 打包发布 V2 版本 |

## 六、风险与安全注意事项

1. **API 密钥和证书绝对不能上传到 GitHub**：`secrets/` 已在 `.gitignore` 中忽略
2. **生产环境建议部署支付回调服务器**：纯CMD轮询可能错过订单状态；回调+轮询双保险更可靠
3. **签名算法要严格按微信文档**：签名错误是最常见的坑
4. **金额单位务必是「分」**：传错单位会导致少收款或多收款（这是高频事故）
5. **订单号唯一**：重复订单号会被微信拒绝
6. **沙箱测试先跑通**：微信支付有沙箱模式，先在沙箱验证再上线
